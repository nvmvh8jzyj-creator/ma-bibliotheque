function openVintedSearch(titreRaw, auteurRaw) { let text = `${titreRaw || ''} ${auteurRaw || ''}`; // 1. Enlever les sous-titres (ex: ": Tome 1" ou ": Illuminer les ombres") text = text.replace(/:.*$/g, ''); // 2. Remplacer les apostrophes, tirets et caractères spéciaux par un espace text = text.replace(/['’\-:_.,]/g, ' '); // 3. Nettoyer les termes "Vol", "Tome", "Livre" et garder un seul espace entre chaque mot text = text.replace(/\b(Vol|Tome|Livre)\b/gi, '') .replace(/\s+/g, ' ') .trim(); // 4. Ajouter le mot livre pour cibler le rayon livres let query = `${text} livre`; // Utilisation d'un encodage URL standard sans signe "+" window.open(`https://www.vinted.fr/catalog?search_text=${encodeURIComponent(query).replace(/%20/g, ' ')}`, '_blank'); }
